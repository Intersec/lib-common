19//! Infrastructure for creating tokio threads able to communicate with C event loop.
20//!
21//! This provides a `TokioCMod` exposed to C with [`c_module!`] as
22//! `tokio_c_module`. A global [`ArcSwapOption`] stores the tokio runtime's
23//! [`Handle`] allowing us to run a thread with a custom [`Future`] object.
24//!
25//! # Overview
26//!
27//! The `tokio_c_module` is implemented with 3 methods:
28//! - `initialize` : initialize the module, with a `shutdown_send` signal and a runtime.
29//! - `on_term` : send the shutdown signal to the Tokio thread when the process
30//!   receives a termination signal.
31//! - `shutdown` : terminate the thread and make sure it is finished.
32//!
33//! The [`spawn`] function allows us to spawn a task using the `tokio_c_module`
34//! runtime and the [`Future`] input. It returns a [`tokio::task::JoinHandle`].
35//!
36//! # Example
37//!
38//! ## Use `tokio_c_module` with `main_c_queue_schedule`
39//! ```
40//! # use tokio_c_mod::tokio_get_module;
41//! # use libcommon_core::bindings::{el_blocker_register, el_loop, el_unregister, ev_t};
42//! # use libcommon_core::module::{module_is_loaded, module_release, module_require};
43//! # use libcommon_core::thr::main_c_queue_schedule;
44//! # struct ElBlocker(*mut ev_t);
45//! # unsafe impl Send for ElBlocker {}
46//! module_require(tokio_get_module());
47//! let blocker = ElBlocker(unsafe { el_blocker_register() });
48//!
49//! tokio_c_mod::spawn(async move {
50//!     main_c_queue_schedule(move || {
51//!         let mut blocker = blocker;
52//!         unsafe {
53//!             el_unregister(&raw mut blocker.0);
54//!         }
55//!     });
56//! });
57//!
58//! unsafe {
59//!     el_loop();
60//! }
61//! module_release(tokio_get_module());
62//! ```
63
64use arc_swap::ArcSwapOption;
65use libcommon_core::bindings::{on_term_method, thr_get_module};
66use libcommon_core::{c_module, thr};
67use std::future::Future;
68use std::os::raw::{c_int, c_void};
69use std::sync::Arc;
70use std::thread::{self, JoinHandle};
71use tokio::runtime::{Builder, Handle};
72use tokio::sync::oneshot::{self, Sender};
73use tokio::task::JoinHandle as TokioJoinHandle;
74
75// {{{ TokioCMod
76
77/// Global variable to store the Tokio C Module runtime handle.
78///
79/// An [`ArcSwapOption`] so the handle can be cleared on shutdown and set
80/// again on a later initialization: the module can be loaded and unloaded
81/// several times (mostly in tests). Readers in [`spawn`] take their own
82/// `Arc`, so a concurrent shutdown storing `None` cannot free a handle still
83/// in use.
84static TOKIO_RUNTIME: ArcSwapOption<Handle> = ArcSwapOption::const_empty();
85
86/// Structure to build a tokio c module.
87///
88/// Fields are Options so the struct can derive Default, since neither
89/// the Sender nor the thread `JoinHandle` exists before initialization.
90#[derive(Default)]
91struct TokioCMod {
92    shutdown_send: Option<Sender<()>>,
93    tokio_thr: Option<JoinHandle<()>>,
94}
95
96/// Spawn a future on the `tokio_c_module` runtime.
97///
98/// # Arguments
99///
100/// `future` - the async task to execute on the `TOKIO_RUNTIME`
101///
102/// # Returns
103///
104/// Returns a [`tokio::task::JoinHandle`] for the spawned future.
105///
106/// # Panics
107///
108/// Panics if the `tokio_c_module` is not loaded, as there is no runtime to
109/// spawn the future on. Require the module before calling [`spawn`].
110///
111/// # Example
112///
113/// ```
114/// # use tokio_c_mod::tokio_get_module;
115/// # use libcommon_core::module::{module_require,module_release};
116/// #
117/// # module_require(tokio_get_module());
118///   tokio_c_mod::spawn(async move {
119///      println!("some task");
120///   });
121/// # module_release(tokio_get_module());
122/// ```
123pub fn spawn<F>(future: F) -> TokioJoinHandle<F::Output>
124where
125    F: Future + Send + 'static,
126    F::Output: Send + 'static,
127{
128    let runtime_opt = TOKIO_RUNTIME.load();
129    let runtime = runtime_opt
130        .as_ref()
131        .expect("tokio_c_module is not initialized");
132    runtime.spawn(future)
133}
134
135// }}}
136// {{{ tokio c module
137
138/// Initialize a `tokio_c_module` and the global variable `TOKIO_RUNTIME`.
139fn tokio_c_mod_initialize(ctx: &mut TokioCMod, _arg: *mut c_void) {
140    // Oneshot because tokio_c_mod must be reinitialized after shutdown, not reused.
141    let (shutdown_send, shutdown_recv) = oneshot::channel();
142    // Oneshot to hand the runtime handle back so we can wait for the tokio
143    // thread to have built its runtime before returning.
144    let (runtime_send, runtime_recv) = oneshot::channel();
145    ctx.shutdown_send = Some(shutdown_send);
146    ctx.tokio_thr = Some(thread::spawn(move || {
147        let runtime = Builder::new_multi_thread()
148            .enable_all()
149            .on_thread_start(thr::attach)
150            .on_thread_stop(thr::detach)
151            .build()
152            .expect("failed to build tokio runtime");
153        runtime_send
154            .send(runtime.handle().clone())
155            .expect("tokio runtime handle receiver dropped");
156
157        let _unused = runtime.block_on(shutdown_recv);
158    }));
159
160    // Wait for the tokio thread to have built its runtime. blocking_recv() is
161    // safe here: initialize() runs on the main C thread, outside any tokio
162    // runtime.
163    let handle = runtime_recv
164        .blocking_recv()
165        .expect("tokio thread stopped before sending its runtime handle");
166    TOKIO_RUNTIME.store(Some(Arc::new(handle)));
167}
168
169/// Send a signal to terminate the tokio c mod thread.
170fn send_shutdown(ctx: &mut TokioCMod) {
171    if let Some(shutdown_send) = ctx.shutdown_send.take() {
172        let _unused: Result<(), ()> = shutdown_send.send(());
173    }
174}
175
176fn tokio_c_mod_on_term(ctx: &mut TokioCMod, _arg: c_int) {
177    send_shutdown(ctx);
178}
179
180/// Shutdown the `tokio_c_module` and terminate the thread.
181fn tokio_c_mod_shutdown(ctx: &mut TokioCMod) {
182    send_shutdown(ctx);
183    if let Some(thread) = ctx.tokio_thr.take() {
184        thread.join().expect("tokio thread panicked");
185    }
186    TOKIO_RUNTIME.store(None);
187}
188
189// `tokio_c_module` module builder. See `module.rs` for further documentation
190c_module!(tokio, TokioCMod, |builder| {
191    builder
192        .depends_on(unsafe { thr_get_module() })
193        .initialize(tokio_c_mod_initialize)
194        .implement_int(&raw const on_term_method, tokio_c_mod_on_term)
195        .shutdown(tokio_c_mod_shutdown);
196});

199#[cfg(test)]
200mod tests {
201
202    use crate::tokio_get_module;
203    use libcommon_core::bindings::{el_blocker_register, el_loop, el_unregister, ev_t};
204    use libcommon_core::module::{module_is_loaded, module_release, module_require};
205    use libcommon_core::thr::main_c_queue_schedule;
206    struct ElBlocker(*mut ev_t);
207    unsafe impl Send for ElBlocker {}
208
209    #[test]
210    fn shutdown_tokio_c_mod_after_c_event_loop() {
211        module_require(tokio_get_module());
212        assert!(module_is_loaded(tokio_get_module()));
213
214        //  Register a blocker so el_loop() keeps running until the spawned
215        //  future fires. The tokio runtime is already live at this point
216        //  (module_require returned), so spawn() succeeds immediately. When
217        //  tokio polls the future it posts el_unregister to the C main queue
218        //  via main_c_queue_schedule, which wakes el_loop. el_loop then
219        //  processes the callback, clears the blocker, and returns. There is
220        //  no race: el_loop cannot exit before the blocker is cleared.
221        let blocker = ElBlocker(unsafe { el_blocker_register() });
222
223        crate::spawn(async move {
224            main_c_queue_schedule(move || {
225                let mut blocker = blocker;
226
227                unsafe {
228                    el_unregister(&raw mut blocker.0);
229                }
230            });
231        });
232
233        unsafe {
234            el_loop();
235        }
236
237        module_release(tokio_get_module());
238        assert!(!module_is_loaded(tokio_get_module()));
239    }
240}
