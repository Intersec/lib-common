179    #[test]
180    fn generic_struct_union_outlives_source_ctx() {
181        // A GenericStructUnion keeps alive (a clone of) the EnvCtx it was
182        // unpacked from, so its descriptor stays valid even after the
183        // source ctx is dropped. Exercises EnvCtx::clone + the dup/release
184        // balance (ASAN-checked).
185        let ctx = setup_env_with_packages();
186        let t_scope = TScope::new_scope();
187
188        let desc = ctx
189            .get_struct_desc("ic.Tracer")
190            .expect("ic.Tracer should exist");
191        let obj = ctx
192            .t_junpack_desc(&t_scope, r#"{"token": 1, "epoch": 2}"#, desc, 0)
193            .expect("valid JSON should unpack");
194
195        // Drop the ctx we resolved `desc` from; `obj` holds its own clone.
196        drop(ctx);
197
198        // `obj`'s descriptor is still valid (read by `as_json`).
199        assert_eq!(obj.as_json(), "{\n    \"token\": 1,\n    \"epoch\": 2\n}\n");
200    }

821    fn dup_generic_outlives_source() {
822        // `dup` on a `GenericStructUnion` yields a lifetime-free `OwnedGeneric`
823        // that clones the `EnvCtx`, so it stays valid after both the source
824        // handle, its `t_scope`, and the ctx it was resolved from are gone.

827            let t_scope = TScope::new_scope();

844        );
