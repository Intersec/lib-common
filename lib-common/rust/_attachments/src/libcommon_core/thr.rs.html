19//! Module to export the thread functions used by `libcommon` in a safe way
20//!
21//! Using `attach()` and `detach()` are required to use `TScope`.
22//!
23//! `main_c_queue_schedule()` can be used to run a closure into the main C thread.
24
25use crate::bindings::{
26    thr_attach, thr_detach, thr_is_initialized, thr_is_on_queue, thr_job_t, thr_queue,
27    thr_queue_main_g, thr_syn_t,
28};
29
30// {{{ Attach/Detach
31
32/// Attach the thread to `libcommon` and call the attach callbacks.
33pub fn attach() {
34    unsafe {
35        thr_attach();
36    };
37}
38
39/// Detach the thread from `libcommon` and call the detach callbacks.
40pub fn detach() {
41    unsafe {
42        thr_detach();
43    };
44}
45
46// }}}
47// {{{ Run main thread
48
49#[repr(C)]
50struct MainCThreadJob<F>
51where
52    F: FnOnce() + Send + 'static,
53{
54    // Should always be put at the beginning to allow proper casting.
55    job: thr_job_t,
56    callback: F,
57}
58
59/// Trampoline function to call the closure from the main C thread.
60unsafe extern "C" fn main_c_thread_run_job<F>(job: *mut thr_job_t, _syn: *mut thr_syn_t)
61where
62    F: FnOnce() + Send + 'static,
63{
64    let job_ptr = job.cast::<MainCThreadJob<F>>();
65    let job = unsafe { Box::from_raw(job_ptr) };
66    (job.callback)();
67}
68
69/// Schedule a closure to be run on the main C thread.
70///
71/// # Example
72///
73/// ```ignore
74/// main_c_queue_schedule(move || {
75///     unsafe {
76///         ic_reply(...);
77///     }
78/// });
79/// ```
80pub fn main_c_queue_schedule<F>(callback: F)
81where
82    F: FnOnce() + Send + 'static,
83{
84    // Assert that the `thr` module is initialized.
85    debug_assert!(unsafe { thr_is_initialized() });
86
87    if unsafe { thr_is_on_queue(thr_queue_main_g) } {
88        // We are already on the main C thread, run the callback immediately.
89        callback();
90        return;
91    }
92
93    // Else, schedule the callback on the main thread.
94    let job = MainCThreadJob {
95        job: thr_job_t {
96            // Bindgen is weird and expects an Option here.
97            run: Some(main_c_thread_run_job::<F>),
98        },
99        callback,
100    };
101
102    let job = Box::into_raw(Box::new(job));
103    unsafe {
104        thr_queue(thr_queue_main_g, job.cast());
105    }
106}

109// {{{ Tests
110
111#[cfg(test)]
112mod test {
113    use std::sync::atomic::{AtomicU32, Ordering};
114    use std::thread;
115
116    use super::{attach, detach, main_c_queue_schedule};
117    use crate::bindings::{
118        el_blocker_register, el_loop, el_unregister, ev_t, module_release, module_require,
119        thr_assert_is_main_thread, thr_get_module,
120    };
121
122    static RESULT: AtomicU32 = AtomicU32::new(0);
123
124    struct ElBlocker(*mut ev_t);
125    unsafe impl Send for ElBlocker {}
126
127    #[test]
128    fn main_c_queue_schedule_from_other_thread() {
129        unsafe {
130            module_require(thr_get_module());
131        }
132
133        let blocker = ElBlocker(unsafe { el_blocker_register() });
134
135        let handle = thread::spawn(move || {
136            attach();
137            main_c_queue_schedule(move || {
138                // Force capture of the whole `blocker` (which is `Send`) instead of
139                // disjointly capturing `blocker.0` (which is not `Send`).
140                let mut blocker = blocker;
141
142                unsafe {
143                    thr_assert_is_main_thread();
144                }
145
146                RESULT.store(42, Ordering::SeqCst);
147
148                unsafe {
149                    el_unregister(&raw mut blocker.0);
150                }
151            });
152            detach();
153        });
154
155        unsafe {
156            el_loop();
157        }
158
159        handle.join().expect("rust thread panicked");
160
161        assert_eq!(RESULT.load(Ordering::SeqCst), 42);
162
163        unsafe {
164            module_release(thr_get_module());
165        }
166    }
167}
