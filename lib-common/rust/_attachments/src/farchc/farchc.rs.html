36const PATHMAX: i32 = 4096;
37const LZO_BUF_MEM_SIZE: usize = 1 << (14 + size_of::<u32>());
38const FARCH_MAX_SYMBOL_SIZE: usize = 128;
39
40struct FarchEntry {
41    pub name: String,
42    pub compressed_size: i32,
43    pub size: i32,
44    pub nb_chunks: i32,
45}

47/// Processes a farch script
48#[derive(Debug, Parser)]
49#[command(name = "farch", about = "Processes a farch script")]
50struct Opts {
51    /// The farch script file to process
52    script: PathBuf,
53
54    /// Output file
55    #[arg(short, long, value_name = "FILE")]
56    out: Option<PathBuf>,
57
58    /// Be verbose
59    #[arg(short, long, action = clap::ArgAction::SetTrue)]
60    verbose: bool,
61}

63fn rand_range(first: usize, last: usize) -> usize {
64    rand::random_range(first..=last)
65}
66
67fn put_as_str(data: &[u8], output: &mut dyn Write) {
68    for byte in data {
69        write!(output, "\\x{byte:x}").unwrap();
70    }
71}
72
73fn put_chunk(chunk: &[u8], output: &mut dyn Write) {
74    write!(output, "    LSTR_IMMED(\"").unwrap();
75    put_as_str(chunk, output);
76    writeln!(output, "\"),").unwrap();
77}
78
79fn obfuscate_data(data: &[u8], key: u64, output: &[u8]) {
80    let src = lstr::from_bytes(data);
81    let dst = lstr::from_bytes(output);
82
83    unsafe {
84        lstr_obfuscate(src.as_raw(), key, dst.as_raw());
85    }
86}
87
88fn dump_and_obfuscate(buf: &[u8], output: &mut dyn Write) -> i32 {
89    let mut nb_chunks: i32 = 0;
90    let mut len = buf.len();
91    let mut start_slice: usize = 0;
92
93    while len > 0 {
94        let chunk_size = rand_range(FARCH_MAX_SYMBOL_SIZE / 2, FARCH_MAX_SYMBOL_SIZE);
95        let chunk_size = min(chunk_size, len);
96        let end_slice: usize = start_slice + chunk_size;
97        let obfuscated_chunk_slice = &buf[start_slice..end_slice];
98        let obfuscated_output_buf = [0u8; FARCH_MAX_SYMBOL_SIZE];
99        let obfuscated_output = &obfuscated_output_buf[0..chunk_size];
100
101        obfuscate_data(obfuscated_chunk_slice, chunk_size as u64, obfuscated_output);
102        put_chunk(obfuscated_output, output);
103
104        start_slice = end_slice;
105        len -= chunk_size;
106        nb_chunks += 1;
107    }
108
109    nb_chunks
110}

112#[allow(clippy::large_stack_arrays)]
113fn dump_file(path: &Path, entry: &mut FarchEntry, output: &mut dyn Write) {
114    let file_data = read(path).unwrap_or_else(|e| {
115        eprintln!("Error: unable to read file `{}`: {}", path.display(), e);
116        exit(1);
117    });
118
119    let file_data_len_i32 = i32::try_from(file_data.len()).unwrap_or_else(|e| {
120        eprintln!(
121            "Error: file is too big for farchc `{}`: {}",
122            path.display(),
123            e
124        );
125        exit(1);
126    });
127
128    entry.size = file_data_len_i32;
129
130    let mut clen = unsafe { lzo_cbuf_size(file_data.len()) };
131    let mut cbuf = vec![0u8; clen];
132    let mut lzo_buf = [0u8; LZO_BUF_MEM_SIZE];
133    let ps: pstream_t = file_data.as_slice().into();
134
135    unsafe {
136        clen = qlzo1x_compress(
137            cbuf.as_mut_ptr().cast::<c_void>(),
138            clen,
139            ps,
140            lzo_buf.as_mut_ptr().cast::<c_void>(),
141        );
142    }
143
144    if clen < file_data.len() {
145        let clen_i32 =
146            i32::try_from(clen).expect("expected error when converting compress length to i32");
147        entry.nb_chunks = dump_and_obfuscate(&cbuf[0..clen], output);
148        entry.compressed_size = clen_i32;
149    } else {
150        entry.nb_chunks = dump_and_obfuscate(&file_data, output);
151        entry.compressed_size = file_data_len_i32;
152    }
153}

155fn dump_entries(archname: &str, entries: &[FarchEntry], output: &mut dyn Write) {
156    let mut chunk = 0;
157
158    for entry in entries {
159        let obfuscated_name = vec![0u8; entry.name.len()];
160
161        obfuscate_data(
162            entry.name.as_bytes(),
163            entry.nb_chunks as u64,
164            &obfuscated_name,
165        );
166        writeln!(output, "/* {{{{{{ {} */", entry.name).unwrap();
167        writeln!(output, "{{").unwrap();
168        write!(output, "    .name = LSTR_IMMED(\"").unwrap();
169        put_as_str(&obfuscated_name, output);
170        writeln!(output, "\"),").unwrap();
171        writeln!(output, "    .chunks = &{archname}_data[{chunk}],").unwrap();
172        writeln!(output, "    .size = {},", entry.size).unwrap();
173        writeln!(output, "    .compressed_size = {},", entry.compressed_size).unwrap();
174        writeln!(output, "    .nb_chunks = {},", entry.nb_chunks).unwrap();
175        writeln!(output, "}},").unwrap();
176        writeln!(output, "/* }}}}}} */").unwrap();
177        chunk += entry.nb_chunks;
178    }
179}

181fn do_work(opts: &Opts, reldir: &Path, output: &mut impl Write) {
182    let content = read_to_string(&opts.script).unwrap_or_else(|e| {
183        eprintln!(
184            "Error: could not read script file `{}`: {}",
185            opts.script.display(),
186            e
187        );
188        exit(1);
189    });
190
191    let mut lines = content.lines();
192    let mut name_opt: Option<&str> = None;
193
194    // Find first non-empty line and non-comment line for the name
195    for line in &mut lines {
196        let trimmed = line.trim();
197        if trimmed.is_empty() || trimmed.starts_with('#') {
198            continue;
199        }
200        name_opt = Some(trimmed);
201        break;
202    }
203
204    let name = name_opt.unwrap_or_else(|| {
205        eprintln!("Error: no variable name specified");
206        exit(1);
207    });
208
209    if opts.verbose {
210        eprintln!("farchc: creating `{name}`");
211    }
212
213    writeln!(output, "/* This file is generated by farchc. */").unwrap();
214    writeln!(output).unwrap();
215    writeln!(output, "#include <lib-common/farch.h>").unwrap();
216    writeln!(output).unwrap();
217    writeln!(output, "static const farch_data_t {name}_data[] = {{").unwrap();
218
219    let mut entries = Vec::<FarchEntry>::new();
220    let mut lineno = 2;
221
222    for line in lines {
223        lineno += 1;
224
225        if line.len() > PATHMAX as usize {
226            eprintln!("Error: line {lineno} is too long");
227            exit(1);
228        }
229
230        let path = line.trim();
231        if path.is_empty() {
232            continue;
233        }
234
235        if path.starts_with('#') {
236            if opts.verbose {
237                eprintln!("farchc: {path}");
238            }
239            continue;
240        }
241
242        let fullpath = reldir.join(path);
243
244        if opts.verbose {
245            eprintln!("farchc: adding `{}` as `{}`", path, fullpath.display());
246        }
247
248        let mut entry = FarchEntry {
249            name: path.to_owned(),
250            size: 0,
251            compressed_size: 0,
252            nb_chunks: 0,
253        };
254
255        writeln!(output, "/* {{{{{{ {path} */").unwrap();
256
257        dump_file(&fullpath, &mut entry, output);
258        writeln!(output, "/* }}}}}} */").unwrap();
259
260        entries.push(entry);
261    }
262
263    writeln!(output, "}};").unwrap();
264    writeln!(output).unwrap();
265    writeln!(output, "static const farch_entry_t {name}[] = {{").unwrap();
266
267    dump_entries(name, &entries, output);
268    writeln!(output, "{{   .name = LSTR_NULL }},").unwrap();
269    writeln!(output, "}};").unwrap();
270}
271
272fn main() {
273    let opts = Opts::parse();
274
275    // Handle input file
276    let script_path = &opts.script;
277
278    let reldir = script_path.parent().unwrap_or_else(|| {
279        eprintln!(
280            "Error: unable to get parent directory of script `{}`",
281            script_path.display()
282        );
283        exit(1);
284    });
285
286    let mut output = String::new();
287
288    do_work(&opts, reldir, &mut output);
289
290    // Output to a file
291    if let Some(out_path) = &opts.out {
292        // Write file
293        let mut out_file = File::create(out_path).unwrap_or_else(|e| {
294            eprintln!(
295                "Error: unable to open `{}` for writing: {}",
296                out_path.display(),
297                e
298            );
299            exit(1);
300        });
301        out_file.write_all(output.as_bytes()).unwrap_or_else(|e| {
302            eprintln!(
303                "Error: unable to write file `{}`: {}",
304                out_path.display(),
305                e
306            );
307            exit(1);
308        });
309
310        // Set permissions on output
311        if let Err(e) = fs::set_permissions(out_path, fs::Permissions::from_mode(0o440)) {
312            eprintln!("Error: unable to chmod `{}`: {}", out_path.display(), e);
313            exit(1);
314        }
315    } else {
316        // Output to stdout
317        io::stdout()
318            .write_all(output.as_bytes())
319            .unwrap_or_else(|e| {
320                eprintln!("Error: unable to write to stdio: {e}");
321                exit(1);
322            });
323    }
324}
